PROG -> eof | (STMT)(STMT)*

STMT -> ([ENTT] iden = EXPR | EXPR);
ENTT -> iden // Temporarily supporting only singleton entities

EXPR -> ASSN(, ASSN)*
ASSN -> TERN = ASSN | TERN
TERN -> LOG ? TERN : TERN | LOG
LOG -> EQ((&& | \|\|) EQ)*
EQ -> INEQ((!= | ==) INEQ)*
INEQ -> ADD((> | < | >= | <=) ADD)*
ADD -> FACT((+ | -) FACT)*
FACT -> UN((\* | /) UN)*
UN -> (! | -)UN | GRP
GRP -> \(COM\) | literal | iden
